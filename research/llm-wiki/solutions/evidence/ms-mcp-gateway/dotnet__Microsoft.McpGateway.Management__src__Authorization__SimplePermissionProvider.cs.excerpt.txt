10:     /// <summary>
11:     /// Simple permission evaluator that allows read access for creators, administrators, or members of a resource's allowed roles,
12:     /// and restricts write access to the resource owner or administrators.
13:     /// </summary>
14:     public class SimplePermissionProvider : IPermissionProvider
15:     {
16:         private const string AdminRole = "mcp.admin";
17: 
18:         public Task<bool> CheckAccessAsync(ClaimsPrincipal principal, IManagedResource resource, Operation operation)
19:         {
20:             ArgumentNullException.ThrowIfNull(principal);
21:             ArgumentNullException.ThrowIfNull(resource);
22: 
23:             var allowed = operation switch
24:             {
25:                 Operation.Read => CanRead(principal, resource),
26:                 Operation.Write => CanWrite(principal, resource),
27:                 _ => false
28:             };
29: 
30:             return Task.FromResult(allowed);
31:         }
32: 
33:         public Task<T[]> CheckAccessAsync<T>(ClaimsPrincipal principal, IEnumerable<T> resources, Operation operation)
34:             where T : IManagedResource
35:         {
36:             ArgumentNullException.ThrowIfNull(principal);
37:             ArgumentNullException.ThrowIfNull(resources);
38: 
39:             var allowedResources = resources
40:                 .Where(resource => resource is not null)
41:                 .Where(resource => operation switch
42:                 {
43:                     Operation.Read => CanRead(principal, resource!),
44:                     Operation.Write => CanWrite(principal, resource!),
45:                     _ => false
46:                 })
47:                 .ToArray();
48: 
49:             return Task.FromResult(allowedResources);
50:         }
51: 
52:         private static bool CanRead(ClaimsPrincipal principal, IManagedResource resource)
53:         {
54:             if (principal == null)
55:             {
56:                 return false;
57:             }
58: 
59:             if (string.Equals(principal.GetUserId(), resource.CreatedBy, StringComparison.OrdinalIgnoreCase))
60:             {
61:                 return true;
62:             }
63: 
64:             var roles = principal.GetUserRoles();
65:             if (IsAdmin(roles))
66:             {
67:                 return true;
68:             }
69: 
70:             if (resource.RequiredRoles == null || resource.RequiredRoles.Count == 0)
71:             {
72:                 return false;
73:             }
74: 
75:             return resource.RequiredRoles.Any(role => roles.Any(userRole => string.Equals(userRole, role, StringComparison.OrdinalIgnoreCase)));
76:         }
77: 
78:         private static bool CanWrite(ClaimsPrincipal principal, IManagedResource resource)
79:         {
80:             if (principal == null)
81:             {
82:                 return false;
83:             }
84: 
85:             if (string.Equals(principal.GetUserId(), resource.CreatedBy, StringComparison.OrdinalIgnoreCase))
86:             {
87:                 return true;
88:             }
89: 
90:             return IsAdmin(principal.GetUserRoles());
91:         }
92: 
93:         private static bool IsAdmin(IEnumerable<string> roles) =>
94:             roles.Any(role => string.Equals(role, AdminRole, StringComparison.OrdinalIgnoreCase));
95:     }
