55:     redis.call('EXPIRE', KEYS[i], window + 1)
56:     if count >= limit then

60:         results[i] = {0, count, oldest_ts}

64:         local oldest = redis.call('ZRANGE', KEYS[i], 0, 0, 'WITHSCORES')
65:         local oldest_ts = 0

102:         local ttl = math.ceil((capacity - tokens) / rate) + 1
103:         redis.call('EXPIRE', KEYS[i], ttl)
